3) Packaging — RPM
3.1 scripts/build-rpm.sh
This helper downloads release artifacts into SOURCES/, verifies them, then calls rpmbuild.

#!/usr/bin/env bash
set -euo pipefail

# Usage: scripts/build-rpm.sh v0.10.0
VERSION="${1:?tag like v0.10.0}"
ARCH="${2:-$(uname -m)}"   # x86_64 or aarch64
REPO="github.com/ORG/REPO"
BASE="https://$REPO/releases/download/$VERSION"

# Map uname -m to Go/release arch suffix
case "$ARCH" in
  x86_64|amd64)   RELARCH=amd64;   RPMARCH=x86_64 ;;
  aarch64|arm64)  RELARCH=arm64;   RPMARCH=aarch64 ;;
  *) echo "Unsupported arch: $ARCH"; exit 2 ;;
 esac

WORKDIR=$(mktemp -d)
trap 'rm -rf "$WORKDIR"' EXIT

pushd "$WORKDIR" >/dev/null
curl -fsSLO "$BASE/nftban-feeds-$RELARCH"
curl -fsSLO "$BASE/nftban-geoip-$RELARCH"
curl -fsSLO "$BASE/SHA256SUMS"
# Try optional signature; do not fail if missing
curl -fsSL -o SHA256SUMS.asc "$BASE/SHA256SUMS.asc" || true

sha256sum -c <(grep "$RELARCH$" SHA256SUMS)  # verifies only lines ending with -$RELARCH

# Prepare rpmbuild tree
RPMTOP=${RPMTOP:-"$HOME/rpmbuild"}
mkdir -p "$RPMTOP"/{SOURCES,SPECS,BUILD,RPMS,SRPMS}
cp nftban-feeds-$RELARCH "$RPMTOP/SOURCES/nftban-feeds"
cp nftban-geoip-$RELARCH "$RPMTOP/SOURCES/nftban-geoip"
cp SHA256SUMS "$RPMTOP/SOURCES/SHA256SUMS"
[ -s SHA256SUMS.asc ] && cp SHA256SUMS.asc "$RPMTOP/SOURCES/SHA256SUMS.asc" || true

5) Local Developer Convenience — Improved Go Build Script
scripts/build-go-binaries.sh

#!/usr/bin/env bash
set -euo pipefail

ARCHES=(amd64 arm64)
OUTDIR=${OUTDIR:-dist}

while [[ $# -gt 0 ]]; do
  case "$1" in
    --arch) ARCHES=("$2"); shift 2 ;;
    --all) ARCHES=(amd64 arm64); shift ;;
    *) echo "Unknown arg: $1"; exit 2 ;;
  esac
done

build_one() {
  local GOARCH=$1
  mkdir -p "$OUTDIR"
  echo "==> Building for linux/$GOARCH"
  GOOS=linux GOARCH=$GOARCH \
  go build -trimpath -ldflags "-s -w" -o "$OUTDIR/nftban-feeds-$GOARCH" ./go-feeds/cmd/nftban-feeds
  GOOS=linux GOARCH=$GOARCH \
  go build -trimpath -ldflags "-s -w" -o "$OUTDIR/nftban-geoip-$GOARCH" ./go-geoip/cmd/nftban-geoip
}

for a in "${ARCHES[@]}"; do build_one "$a"; done

( cd "$OUTDIR" && rm -f SHA256SUMS && for f in nftban-*; do sha256sum "$f" >> SHA256SUMS; done )

echo "Binaries in $OUTDIR; checksums in $OUTDIR/SHA256SUMS"

#!/usr/bin/env bash
set -euo pipefail
VERSION="${1:?tag like v0.10.0}"
ARCH="${2:-$(uname -m)}"   # x86_64 or aarch64
REPO="github.com/itcmsgr/nftban"
BASE="https://$REPO/releases/download/$VERSION"
case "$ARCH" in
  x86_64|amd64)   RELARCH=amd64;   RPMARCH=x86_64 ;;
  aarch64|arm64)  RELARCH=arm64;   RPMARCH=aarch64 ;;
  *) echo "Unsupported arch: $ARCH"; exit 2 ;;
 esac
WORKDIR=$(mktemp -d)
trap 'rm -rf "$WORKDIR"' EXIT
pushd "$WORKDIR" >/dev/null
curl -fsSLO "$BASE/nftban-feeds-$RELARCH"
curl -fsSLO "$BASE/nftban-geoip-$RELARCH"
curl -fsSLO "$BASE/SHA256SUMS"
curl -fsSL -o SHA256SUMS.asc "$BASE/SHA256SUMS.asc"
gpg --batch --verify SHA256SUMS.asc SHA256SUMS
sha256sum -c <(grep "$RELARCH$" SHA256SUMS)
RPMTOP=${RPMTOP:-"$HOME/rpmbuild"}
mkdir -p "$RPMTOP"/{SOURCES,SPECS,BUILD,RPMS,SRPMS}
cp nftban-feeds-$RELARCH "$RPMTOP/SOURCES/nftban-feeds"
cp nftban-geoip-$RELARCH "$RPMTOP/SOURCES/nftban-geoip"
cp SHA256SUMS "$RPMTOP/SOURCES/SHA256SUMS"
cp SHA256SUMS.asc "$RPMTOP/SOURCES/SHA256SUMS.asc"
repo_root=$(git rev-parse --show-toplevel 2>/dev/null || pwd)
cp "$repo_root/packaging/rpm/nftban.spec" "$RPMTOP/SPECS/"
rpmbuild --define "_topdir $RPMTOP" \
  --define "_version ${VERSION#v}" \
  --define "_relarch $RELARCH" \
  --define "_rpmarch $RPMARCH" \
  -bb "$RPMTOP/SPECS/nftban.spec"
popd >/dev/null
